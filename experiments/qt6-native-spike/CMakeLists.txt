cmake_minimum_required(VERSION 3.24)
project(JAQtSpike VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick Qml QuickControls2 Multimedia)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(ja-native-spike main.cpp)
qt_add_qml_module(ja-native-spike URI JA.Spike VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(ja-native-spike PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::QuickControls2 Qt6::Multimedia)
set_target_properties(ja-native-spike PROPERTIES WIN32_EXECUTABLE FALSE MACOSX_BUNDLE FALSE)
enable_testing()
add_test(NAME native_contract COMMAND ja-native-spike --self-test)
set_tests_properties(native_contract PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QSG_RHI_BACKEND=software")
